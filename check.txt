# Diagnostic: who sits on shared coordinates
t0 = time.time()
cg = spark.table(OUT_TABLES["cust"]).where(F.col("lat").isNotNull())
sz = F.col("coloc_cluster_size")
is_org = F.col("party_type") == "O"
org_tm = is_org & F.col("is_tm")
BUCKETS = ["1", "2–4", "5–9", "10–24", "25–99", "100–999", "1000+"]
def bucket(c):
    return (F.when(c < 2, "1").when(c < 5, "2–4").when(c < 10, "5–9").when(c < 25, "10–24")
             .when(c < 100, "25–99").when(c < 1000, "100–999").otherwise("1000+"))

# 1) customers by size of their coordinate cluster (size counts every party in the snapshot)
D1 = (cg.groupBy(bucket(sz).alias("cluster_size"))
        .agg(F.count("*").alias("customers"), F.sum(is_org.cast("int")).alias("orgs"),
             F.sum(org_tm.cast("int")).alias("tm_orgs"))
        .toPandas())
D1["_o"] = D1.cluster_size.map(BUCKETS.index); D1 = D1.sort_values("_o").drop(columns="_o")

# 2) geocode record type vs sharing
D2 = (cg.groupBy(F.coalesce("addr_loc_rec_type", F.lit("(null)")).alias("addr_loc_rec_type"))
        .agg(F.count("*").alias("customers"), F.sum((sz >= COLOC_MIN).cast("int")).alias("shared_ge5"),
             F.avg("coord_decimals").alias("avg_decimals"))
        .orderBy(F.desc("customers")).limit(20).toPandas())
D2["shared_share"] = D2.shared_ge5 / D2.customers

# 3) shared points: how many carry several ZIP5s (centroid fallback signal)
pt = (cg.withColumn("pt", F.concat_ws("_", F.round("lat", 5).cast("string"), F.round("lon", 5).cast("string")))
        .groupBy("pt").agg(F.max(sz).alias("parties_all"), F.count("*").alias("customers"),
                           F.sum(org_tm.cast("int")).alias("tm_orgs"), F.min("mdm_id").alias("mdm_id"),
                           F.first("city").alias("city"), F.first("state").alias("state"),
                           F.first("zip5").alias("zip5"), F.countDistinct("zip5").alias("n_zip5"),
                           F.first("addr_loc_rec_type").alias("rec_type"))
        .where(F.col("parties_all") >= COLOC_MIN))
pt.write.mode("overwrite").parquet(f"{WORK}/diag_pt"); pt = spark.read.parquet(f"{WORK}/diag_pt")
D3 = (pt.groupBy(bucket(F.col("parties_all")).alias("cluster_size"),
                 F.when(F.col("n_zip5") > 1, "several ZIP5").otherwise("one ZIP5").alias("zip5_on_point"))
        .agg(F.count("*").alias("points"), F.sum("customers").alias("customers"), F.sum("tm_orgs").alias("tm_orgs"))
        .toPandas())
D3["_o"] = D3.cluster_size.map(BUCKETS.index); D3 = D3.sort_values(["_o", "zip5_on_point"]).drop(columns="_o")

# 4) the 25 largest points, and 25 random small points (5–24) holding a TM organisation, with one address line
ad = (spark.table(T_ADDR).where(F.col("load_dt").cast("string") == SNAP)
        .select(F.trim(F.col("mdm_id")).alias("mdm_id"), "addr_line_1", "addr_line_2"))
def with_addr(df):
    return df.join(ad, "mdm_id", "left").dropDuplicates(["pt"]).drop("mdm_id").toPandas()
COLS = ["parties_all", "customers", "tm_orgs", "city", "state", "zip5", "n_zip5", "rec_type", "addr_line_1", "addr_line_2"]
D4 = with_addr(pt.orderBy(F.desc("parties_all")).limit(25)).sort_values("parties_all", ascending=False)[COLS]
D5 = with_addr(pt.where((F.col("parties_all") < 25) & (F.col("tm_orgs") > 0)).orderBy(F.rand(SEED)).limit(25))[COLS]
delete(f"{WORK}/diag_pt")

out(sec(f"Shared coordinates · {(time.time() - t0) / 60:.1f} min"),
    sec("1) Customers by cluster size"), tbl(D1),
    sec("2) Geocode record type"), tbl(D2, pct=("shared_share",), dec=("avg_decimals",)),
    sec("3) Shared points by size and ZIP5 count"), tbl(D3),
    sec("4) Largest 25 points"), tbl(D4),
    sec("5) Random small points (5–24) with a TM organisation"), tbl(D5))
